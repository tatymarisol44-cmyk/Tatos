from pathlib import Path
import subprocess, os, re, json, sys, time

repo=Path(r'C:\Users\momoy\Desktop\VS CODE\AGENTES\agency-orchestrator')
out=Path(__file__).resolve().parent/'agency-audit-2026-10-19'
name=sys.argv[1]
env=os.environ.copy()
for field in re.findall(r'^    ([a-z][a-z0-9_]*):', (repo/'src/orchestrator/config.py').read_text(),re.M):
    env.pop(field.upper(),None)
for key in list(env):
    if key.startswith('TEST_') or key.endswith(('_API_KEY','_API_TOKEN','_ACCESS_TOKEN')):
        env.pop(key,None)
env['PYTHONPATH']=str(repo/'src')
env['COVERAGE_FILE']=str(out/(name+'.coverage'))
env['PATH']=str(Path(__file__).resolve().parent/'node-v24.21.0-win-x64')+os.pathsep+env['PATH']
if name.startswith(('postgres-', 'final-')):
    state=json.loads((out/'postgres-state.json').read_text())
    assert '@127.0.0.1:' in state['test_postgres_url']
    env['TEST_POSTGRES_URL']=state['test_postgres_url']
    env['PG_BIN']=state['pg_bin']
    env['PATH']=str(out/'tooling')+os.pathsep+r'C:\Program Files\Git\bin'+os.pathsep+env['PATH']
args=[str(repo/'.venv/Scripts/python.exe'),'-m','pytest','-q',
      '--basetemp='+str(out/('tmp-'+name)),
      '-o','cache_dir='+str(out/('cache-'+name)),
      '--junitxml='+str(out/(name+'.junit.xml')),*sys.argv[2:]]
print(json.dumps({'phase':'TESTS_STARTED','name':name,'source':'existing working tree including preserved changes','live_provider_environment_removed':True}),flush=True)
start=time.monotonic()
with (out/(name+'.log')).open('w',encoding='utf-8') as log:
    result=subprocess.run(args,cwd=repo,env=env,stdout=log,stderr=subprocess.STDOUT)
summary={'name':name,'exit_code':result.returncode,'elapsed_seconds':round(time.monotonic()-start,2)}
(out/(name+'.result.json')).write_text(json.dumps(summary,indent=2))
print(json.dumps(summary),flush=True)
print((out/(name+'.log')).read_text(encoding='utf-8')[-6500:],flush=True)
raise SystemExit(result.returncode)
